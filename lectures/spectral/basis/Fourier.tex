\section{Fourier series}

\subsection{Continuous Fourier series}
The set of functions, for $k \in \mathbb Z$,
\begin{equation*}
    \phi_k(x) = e^{ikx}
\end{equation*}
is an orthogonal system over the interval $(0,2\pi)$ since
\begin{equation*}
    \int_0^{2\pi} \phi_k \overline \phi_l = 2\pi \delta_{kl}.
\end{equation*}
For $u : (0,2\pi) \to \mathbb C$ we introduce the Fourier coefficients
\begin{equation*}
    \hat u_k = \frac 1 {2\pi} \int_0^{2\pi} u(x) \overline \phi_k(x) \diff x.
\end{equation*}
The Fourier series is defined as
\begin{equation*}
    Su = \sum_{k=-\infty}^\infty \hat u_k \phi_k.
\end{equation*}
As an approximation we use, for $N$ even, the trigonometric polynomials
\begin{equation*}
    P_N u = \sum_{k=-N/2}^{N/2-1} \hat u_k \phi_k.
\end{equation*}
Recall that if $u$ is periodic and $C^1$ (or, more generally, Hölder continuous), then $P_N u \to u$ uniformly in $(0,2\pi)$. Continuity alone is not enough for uniform convergence.
We also define the space
\begin{equation*}
    S_N = \operatorname{span}_{\mathbb C} \{ \phi_k : -N/2 \le k \le N/2 - 1\}.
\end{equation*}
We recall Parseval's identity
\begin{equation*}
    \| u \|_{L^2(0,2\pi)}^2 = 2 \pi \sum_{k=-\infty}^\infty |\hat u_k|^2.
\end{equation*}
Thus
\begin{equation*}
    u = \sum_{k=-\infty}^\infty \hat u_k \phi_k \qquad \forall u \in L^2(0,2\pi).
\end{equation*}

\begin{example}
    Let
    \begin{equation*}
        u(x) = \begin{dcases}
            1 & \text{if } \frac \pi 2 < x \le \frac {3\pi}{2} \\
            0 & \text{otherwise}.
        \end{dcases}
    \end{equation*}
    Then
    \begin{equation*}
        \hat u_k = \begin{dcases}
            \frac 1 2                     & \text{if } k = 0                \\
            0                             & \text{if } k \ne 0 \text{ even} \\
            -\frac{(-1)^{(k-1)/2}}{k \pi} & \text{if } k \text{ odd.}
        \end{dcases}
    \end{equation*}
\end{example}
If $u: (0,2\pi)\to \mathbb R$ is real-valued then $\hat u_{-k} = \overline{\hat u_k}$, and we can expand the complex exponentials as trigonometric functions
\begin{equation*}
    u(x) = \sum_{k=-\infty}^{\infty} a_k \cos k x + \sum_{k=-\infty}^\infty b_k \sin k x,
\end{equation*}
where $a_k = \operatorname{Re} \hat u_k$ and $b_k = -\operatorname{Im} \hat u_k$, i.e. $\hat u_k = a_k - i b_k$. Notice that, due to the asymmetric truncation, $P_N u$ need not be real-valued (the term $k = -N/2$ has no conjugate partner); taking the real part changes the approximation space. The general Fourier formulas below use the complex space $S_N$. For real-valued approximations one can instead use symmetric frequencies (for example, an odd number of modes).

\subsection{Periodic functions}

The Fourier series are specially useful for the periodic setting. Let us define
\begin{equation*}
    C^m_p([a,b]) = \left\{ u\in C^m ([a,b]) : \frac{d^k u}{dx^k}(a) = \frac{d^k u}{dx^k}(b) \text{ for } 0 \le k \le m \right\}.
\end{equation*}
We define
\begin{equation*}
    H^m_p(a,b) = \overline{C^m_p([a,b])}^{H^m(a,b)}.
\end{equation*}
We introduce
\begin{equation*}
    \newnorm{u}_{H_p^m(0,2\pi)} = \left(\sum_{k=-\infty}^{\infty} (1 + |k|^{2m}) |\hat u_k|^2 \right)^{\frac 1 2}.
\end{equation*}
This is an equivalent norm of $H_p^m(0,2\pi)$.
\begin{remark}
    Notice that $L^2_p = L^2$, and hence we do not need to assume that $u$ is periodic for $P_N u$ to converge in $L^2(0,2\pi)$.
\end{remark}


\subsection{Discrete Fourier series}
For pointwise sampling assume $u\in C_p([0,2\pi])$. Let us take
\begin{equation*}
    x_j = \frac{2\pi j}{N} \qquad j \in \{0,\dots, N-1\}.
\end{equation*}
We define the discrete Fourier coefficients
\begin{equation*}
    \tilde u_k = \frac 1 N \sum_{j=0}^{N-1} u(x_j) \overline \phi_k(x_j), \qquad k \in \{-N/2, \dots, N/2-1\}.
\end{equation*}
We have the discrete orthogonality relation
\begin{equation*}
    \frac 1 N \sum_{j=0}^{N-1}  \phi_l(x_j) \overline \phi_k(x_j) = \frac 1 N \sum_{j=0}^{N-1} e^{i(l - k)x_j} = \begin{dcases}
        1 & l - k \in N \mathbb Z , \\
        0 & \text{otherwise}.
    \end{dcases}
\end{equation*}
Thus, we have that
\begin{equation*}
    u(x_j) = \sum_{k=-N/2}^{N/2-1} \tilde u_k \phi_k(x_j), \qquad j \in \{0, \dots, N-1\}.
\end{equation*}
We set
\begin{equation*}
    I_N u (x) = \sum_{j=0}^{N-1} u(x_j) \psi_j (x), \qquad \text{where } \psi_j (x) = \frac 1 N \sum_{k=-N/2}^{N/2 - 1} \phi_k(x-x_j).
\end{equation*}
We point out that $\psi_j (x_l) = \delta_{lj}$.

\bigskip

For the following aliasing identity assume, more specifically, $u\in H_p^m(0,2\pi)$ with integer $m\ge1$. Its Fourier coefficients are then absolutely summable. We define the \emph{aliasing error}
\begin{equation*}
    R_N u = \sum_{k=-N/2}^{N/2-1} \phi_k \sum_{\substack{m =-\infty \\ m \ne 0}}^{\infty} \hat u_{k + Nm}.
\end{equation*}
Since $\tilde u_k = \sum_{m \in \mathbb Z} \hat u_{k + Nm}$ (by the discrete orthogonality relation), we have that $I_N u = P_N u + R_N u$ and, by orthogonality,
\begin{equation}
    \|u - I_N u\|^2_{L^2(0,2\pi)} = \|u - P_N u\|^2_{L^2(0,2\pi)} + \|R_N u\|^2_{L^2(0,2\pi)}.
\end{equation}

\subsection{Differentiation}
We define the Fourier interpolation derivative of $u$
\begin{equation*}
    \mathcal D_N u \defeq \frac{\diff}{\diff x} (I_N u).
\end{equation*}
It does not coincide, in general, with $P_N \frac{\diff u}{\diff x}$ or $I_N \frac{\diff u}{\diff x}$.

\subsection{Truncation errors}

\begin{theorem}[Truncation of the Fourier series for a periodic function]
    \label{thm:Fourier truncation of periodic function}
    For integers $0\le k\le m$ and $u\in H_p^m(0,2\pi)$, we have
    \begin{equation*}
        \|u - P_N u\|_{H^k(0,2\pi)} \le C N^{k-m} \|u^{(m)}\|_{L^2(0,2\pi)}.
    \end{equation*}
\end{theorem}
\begin{proof}
    We prove only the case $k = 0$ for simplicity. Using Parseval's identity, and since $|k| \ge N/2$ for the indices not in the truncation,
    \begin{align*}
        \frac 1 {{2\pi}}  \|u - P_N u\|_{L^2(0,2\pi)}^2
         & =
        \sum_{k \notin [-N/2, N/2 - 1]} |\hat{u}_k|^2
        = \sum_{k \notin [-N/2, N/2 - 1]} \frac{1}{|k|^{2m}} |k|^{2m} |\hat{u}_k|^2
        \\
         & \leq \left( \frac N 2 \right)^{-2m} \sum_{k \in \mathbb Z} |k|^{2m} |\hat{u}_k|^2 = \frac{2^{2m}}{2 \pi} N^{-2m} \|u^{(m)}\|_{L^2(0,2\pi)}^2 .\qedhere
    \end{align*}
\end{proof}
As a consequence
\begin{equation*}
    \inf_{\phi \in S_N} \|u - \phi\|_{H^k(0,2\pi)} \le C N^{k-m} \|u^{(m)}\|_{L^2(0,2\pi)}, \qquad \forall u \in H^{m}_p(0,2\pi).
\end{equation*}
For integer $m\ge1$ (so that point values are defined), we can also prove that
\begin{equation*}
    \|R_N u\|_{L^2} \le C N^{-m} \|u^{(m)} \|_{L^2(0,2\pi)}, \qquad u \in H^{m}_p(0,2\pi).
\end{equation*}

Now is a good time for an example.
\begin{example}[Laplace equation with periodic conditions via Fourier--Galerkin]
    \label{ex:Laplace equation by Fourier-Galerkin}
    For real-valued $f\in L^2(0,2\pi;\mathbb R)$, seek a real-valued solution of
    \begin{equation*}
        \begin{dcases}
            -\Delta u = f & \text{in } (0,2\pi), \\
            u(0) = u(2\pi), \quad u'(0) = u'(2\pi).
        \end{dcases}
    \end{equation*}
    We need to assume the compatibility condition $\int_0^{2\pi} f = 0$. Notice that this is exactly $\hat f_0 = 0$.
    This problem is under-determined since for any solution $u$ and any constant $c \in \mathbb R$ we have that $u + c$ is also a solution.
    So we look for the solution with $\int_0^{2\pi} u = 0$.
    We retain complex Fourier coefficients as notation for real functions: $\hat f_{-k}=\overline{\hat f_k}$ and $\hat u_{-k}=\overline{\hat u_k}$.
    Thus, we consider
    \begin{equation*}
        V = \left\{ u \in H^1_p(0,2\pi;\mathbb R) : \int_0^{2\pi} u = 0 \right\}.
    \end{equation*}
    The problem can be written equivalently as finding $u \in V$ such that
    \begin{equation*}
        \int_0^{2\pi}u'\phi'\,\diff x = \int_0^{2\pi}f\phi\,\diff x, \qquad \forall \phi \in V.
    \end{equation*}
    Testing separately with $\cos(kx)$ and $\sin(kx)$, and combining the two real identities, gives
    \begin{equation*}
        k^2 \hat u_k = \hat f_k, \qquad k\ne0.
    \end{equation*}
    The missing equation for the constant mode is the compatibility condition $\hat f_0=0$, obtained by integrating the original equation.
    We can see that the exact solution is
    \begin{equation*}
        u(x) = \sum_{\substack{k=-\infty \\ k \ne 0}}^\infty \frac{\hat f_k}{k^2} \phi_k(x)
        = 2\operatorname{Re}\sum_{k=1}^\infty\frac{\hat f_k}{k^2}e^{ikx}.
    \end{equation*}
    To retain real-valued approximations, choose $N=2K+1$ with $K\ge1$ and use the symmetric real space
    \begin{equation*}
        S_N^{\mathbb R}=\operatorname{span}_{\mathbb R}\{1,\cos x,\sin x,\dots,\cos(Kx),\sin(Kx)\}.
    \end{equation*}
    Its $L^2$-orthogonal projection on real functions is $Q_Nu=\sum_{k=-K}^K\hat u_k\phi_k$. Conjugate pairs make this sum real. The Galerkin problem is to find $u_N\in V_N:=S_N^{\mathbb R}\cap V$ such that
    \begin{equation*}
        \int_0^{2\pi}u_N'\phi'\,\diff x=\int_0^{2\pi}f\phi\,\diff x,
        \qquad\forall\phi\in V_N.
    \end{equation*}
    Hence
    \begin{equation*}
        u_N(x)=\sum_{0<|k|\le K}\frac{\hat f_k}{k^2}\phi_k(x)
        =2\operatorname{Re}\sum_{k=1}^K\frac{\hat f_k}{k^2}e^{ikx}=Q_Nu(x).
    \end{equation*}
    The proof of \Cref{thm:Fourier truncation of periodic function} applies to $Q_N$ with the same powers of $N$, since the omitted frequencies satisfy $|k|\ge K+1>N/2$.
    Higher orders of convergence require $f \in H^m_p(0,2\pi)$.
\end{example}

\begin{remark}[\Cref{ex:Laplace equation by Fourier-Galerkin} in terms of Céa's lemma]
    Let us define for $u,v$ periodic
    \begin{equation*}
        a(u,v) = \int_0^{2\pi} \frac{\diff u}{\diff x}\frac{\diff v}{\diff x}\,\diff x.
    \end{equation*}
    Due to the Poincaré--Wirtinger inequality this is a continuous, coercive, symmetric bilinear form on the real Hilbert space $V$ above. The weak formulation is given by \eqref{eq:Galerkin elliptic variational}, so the real versions of Lax--Milgram and Céa's lemma apply directly.
    Let $h=N^{-1}$ and $V_h=V_N=V\cap S_N^{\mathbb R}$.
    Let $u_h \in V_h$ solve \eqref{eq:Galerkin elliptic variational discrete}.
    Again notice that, since this is a very simple example, $u_h = u_N$.
    Then, due to \namedCref{lem:cea} we have that
    \begin{align*}
        \|u - u_h\|_{H^1(0,2\pi)} & \le C \|u - Q_Nu\|_{H^1(0,2\pi)} \le C h^{(m+2)-1} \|\tfrac{\diff^{m+2} u}{\diff x^{m+2}}\|_{L^2(0,2\pi)} \\
                                  & = C h^{m+1} \| \tfrac{\diff^m f}{\diff x^m} \|_{L^2(0,2\pi)},
    \end{align*}
    whenever $f \in H^m_p(0,2\pi)$.
    And by \Cref{lem:Aubin-Nietsche} (with the real mean-zero pivot space $H=L^2_0(0,2\pi;\mathbb R)$ and $\omega(h)=Ch$) we have also that
    \begin{align*}
        \|u - u_h\|_{L^2(0,2\pi)} & \le C h^{2} \| f \|_{L^2(0,2\pi)}.
    \end{align*}
\end{remark}

    We will discuss numerical integration in the next class.

Continuous mean zero alone does not imply this. If necessary replace the samples by $f_j^0=f(x_j)-N^{-1}\sum_l f(x_l)$, explicitly solving for the projected load.     Compare the numerical errors as a function of $N$.
\end{exercises}
