\chapter{Convergence of spectral methods in some examples}
\section{Chebyshev collocation for Laplace's equation}
We now complete the details on \Cref{ex:collocation Laplace Chebyshev}. Convergence can be proven through consistency.
The proof of consistency is, however, quite unpleasant.
Let us now prove stability.
Consider the Chebyshev weight $w(x) = (1-x^2)^{-\frac 1 2}$ and the weights in \namedCref{ex:Chebyshev Gauss-Lobatto}.
Multiplying each equation by $u_N (x_k) w_k$ (notice that $u_N(x_0) = u_N(x_N) = 0$) and adding we have that
\begin{equation*}
    -\sum_{k=0}^N \frac{d^2 u_N}{dx^2} (x_k) u_N (x_k) w_k = \sum_{k=0}^N f(x_k) u_N(x_k) w_k.
\end{equation*}
We point out that $\frac{\diff^2u_N}{\diff x^2} u_N \in \mathbb R_{2N - 2}[x]$. Therefore, by \Cref{thm:Gauss-Lobatto integration Jacobi weights} we have that
\begin{equation*}
    -\int_{-1}^1 \frac{\diff^2u_N}{\diff x^2} (x) u_N(x) w(x) \diff x = \sum_{k=0}^N f(x_k) u_N(x_k) w_k.
\end{equation*}
It can be proven (see \cite[Section 7.1]{CanutoHussainiQuarteroniZang2011}) that, since $u_N(\pm 1) = 0$,
\begin{equation*}
    -\int_{-1}^1 \frac{\diff^2u_N}{\diff x^2} (x) u_N(x) w(x) \diff x \ge \frac 1 4 \int_{-1}^1 \left| \frac{\diff u_N}{\diff x} (x) \right|^2 w(x) \diff x \ge \frac 1 4 \int_{-1}^1 \left| \frac{\diff u_N}{\diff x} (x) \right|^2 \diff x.
\end{equation*}
Since the weights are non-negative and $\sum_{k=0}^N w_k = \int_{-1}^1 w = \pi$, we have that
\begin{align*}
    \frac 1 4 \int_{-1}^1 \left| \frac{\diff u_N}{\diff x} (x) \right|^2 \diff x
     & \le \sum_{k=0}^N f(x_k) u_N(x_k) w_k                        \\
     & \le \| f \|_{L^\infty(-1,1)} \| u_N \|_{L^\infty(-1,1)} \sum_{k=0}^N w_k   \\
     & = \pi \| f \|_{L^\infty(-1,1)} \| u_N \|_{L^\infty(-1,1)}.
\end{align*}
Since $u_N(-1) = 0$, we have $|u_N(x)| = |\int_{-1}^x \frac{\diff u_N}{\diff x}| \le \sqrt 2 \|\frac{\diff u_N}{\diff x}\|_{L^2(-1,1)}$, and hence, using also Poincaré's inequality,
\begin{equation*}
    \| u_N \|_{H^1(-1,1)} \le C \| f \|_{L^\infty(-1,1)}.
\end{equation*}

\section{Spectral approximation of Burgers' Equation}
We follow \cite[Section 3.3]{CanutoHussainiQuarteroniZang2011}.
Consider the viscous Burgers' equation with periodic boundary conditions
\begin{equation}
    \frac{\partial u}{\partial t} + u \frac{\partial u}{\partial x} = \nu \frac{\partial^2 u }{\partial x^2}.
\end{equation}

\subsection{Fourier--Galerkin in $(0,2\pi)$ with periodic boundary conditions}

Let us look for solutions
\begin{equation}
    \label{eq:Fourier discrete solution}
    u_N(t,x) = \sum_{k=-N/2}^{N/2-1} \hat u_k(t) e^{ikx}.
\end{equation}
The periodic boundary conditions are already imposed in the choice of basis.
We enforce the weak formulation
\begin{equation*}
    \int_0^{2\pi} \left( \frac{\partial u_N}{\partial t} + u_N \frac{\partial u_N}{\partial x} - \nu \frac{\partial^2 u_N}{\partial x^2} \right) e^{-ikx} = 0,
    \qquad \text{for }k = -\frac N 2, \dots, \frac N 2 - 1.
\end{equation*}
Due to orthogonality we have that
\begin{equation*}
    \frac{d \hat u_k}{dt} + \widehat {\left( u_N \frac{\partial u_N}{\partial x} \right)}_k+ k^2 \nu \hat u_k = 0
    \qquad \text{for }k = -\frac N 2, \dots, \frac N 2 - 1
\end{equation*}
where
\begin{equation*}
    \widehat {\left( u_N \frac{\partial u_N}{\partial x} \right)}_k = \frac 1 {2\pi} \int_0^{2\pi} u_N \frac{\partial u_N}{\partial x} e^{-ikx} dx.
\end{equation*}
We observe that
\begin{equation*}
    \widehat {\left( uv \right)}_k = \frac 1 {2\pi} \int_0^{2\pi} uv e^{-ikx} dx = \sum_{\substack{p+q = k \\ -\frac{N}{2} \le p,q \le \frac{N}{2} - 1}} \hat u_p \hat v_q.
\end{equation*}
Therefore we can write the closed (but unpleasant) formula for the drift term
\begin{equation*}
    \widehat {\left( u_N \frac{\partial u_N}{\partial x} \right)}_k = \sum_{\substack{p+q = k \\ -\frac{N}{2} \le p,q \le \frac{N}{2} - 1}} i q \, \hat u_p(t) \hat u_q(t).
\end{equation*}
This kind of sums can be computed reasonably efficiently (see \cite[Section 3.4]{CanutoHussainiQuarteroniZang2011}).

\subsection{Fourier collocation in $(0,2\pi)$ with periodic boundary conditions}
Now we set the collocation formulation on $x_j = 2 \pi j / N$ for $j = 0, \dots, N-1$, $u_N$ given in Fourier expansion \eqref{eq:Fourier discrete solution}
and set the conditions
\begin{equation*}
    \left( \frac{\partial u_N}{\partial t} + u_N \frac{\partial u_N}{\partial x} - \nu \frac{\partial^2 u_N}{\partial x^2} \right)\Bigg|_{x = x_j} = 0 , \qquad \forall t > 0, j \in \{0,\dots, N-1\}.
\end{equation*}
\begin{remark}
    \label{rem:collocation solve in point-values}
    In previous examples like \namedCref{ex:collocation Laplace Chebyshev} it was easier to write the algebraic system of the collocation problem directly in the coefficients with respect to the basis, $\hat u_k(t)$.
    However, due to the multiplication $u_N \frac{\partial u_N}{\partial x}$ this is now not easy. In this case it is therefore better to pass to a system in terms of $u_N(t, x_k)$.
\end{remark}
We take advantage of the notation in \Cref{sec:spectral collocation linear algebra}\footnote{Notice that the Fourier system has $N$ coefficients, whereas in \Cref{sec:spectral collocation linear algebra} we worked with $N+1$.}
to write
\begin{equation*}
    \begin{pmatrix}
        \frac{\partial u_N}{\partial x}(t,x_0) \\ \vdots \\ \frac{\partial u_N}{\partial x} (t, x_{N-1})
    \end{pmatrix}
    =
    \mat D_{N-1}
    \begin{pmatrix}
        u_N(t,x_0) \\ \vdots \\ u_N(t,x_{N-1})
    \end{pmatrix},
    \qquad
    \begin{pmatrix}
        \frac{\partial^2 u_N}{\partial x^2}(t,x_0) \\ \vdots \\ \frac{\partial^2 u_N}{\partial x^2} (t, x_{N-1})
    \end{pmatrix}
    =
    \mat D_{N-1}^2
    \begin{pmatrix}
        u_N(t,x_0) \\ \vdots \\ u_N(t,x_{N-1})
    \end{pmatrix}.
\end{equation*}
Recall that the periodic boundary conditions are already built into the basis functions.
Notice by periodicity that, in this case, $u_N(t, x_N) = u_N(t, x_0)$.
Then letting $\vec u(t) = (u_N(t,x_0), \dots, u_N(t,x_{N-1}))^t$ we recover the system of ODEs
\begin{equation*}
    \frac{\diff \vec u}{\diff t} + \vec u \boxtimes  {\mat D}_{N-1} \vec u - \nu  {\mat D}^2_{N-1} \vec u = 0,
\end{equation*}
where $\boxtimes$ denotes the entry-wise product of vectors
\begin{equation*}
    \begin{pmatrix}
        u_1 \\ \vdots \\ u_N
    \end{pmatrix}
    \boxtimes
    \begin{pmatrix}
        v_1 \\ \vdots \\ v_N
    \end{pmatrix}
    =
    \begin{pmatrix}
        u_1v_1 \\ \vdots \\ u_Nv_N
    \end{pmatrix}.
\end{equation*}
We recall \Cref{rem:pointwise evaluation plotting}.


\subsection{Chebyshev collocation in $(-1,1)$ with Dirichlet boundary conditions}
We can use the Chebyshev--Gauss--Lobatto nodes $x_j = \cos \pi j / N$ for $j = 0, \dots, N$. Then we set
\begin{equation*}
    \left( \frac{\partial u_N}{\partial t} + u_N \frac{\partial u_N}{\partial x} - \nu \frac{\partial^2 u_N}{\partial x^2} \right)\Bigg|_{x = x_j} = 0 , \qquad \forall t > 0, j \in \{1,\dots, N-1\},
\end{equation*}
and
\begin{equation*}
    u_N(t, x_0) = u_L(t), \qquad u_N(t, x_N) = u_R(t),
\end{equation*}
where $u_L$ and $u_R$ are known functions.
Again we have to point out \Cref{rem:collocation solve in point-values}.
For convenience and consistency with \Cref{sec:spectral collocation linear algebra} we write the whole vector $\vec u (t) = (u_L(t), u_N(t, x_1), \dots, u_N(t, x_{N-1}), u_R(t))^t$.
We only have ODEs for the nodes with $j \in \{1, \dots, N-1\}$, so we can write the system as
\begin{equation*}
    \mat Z_N
    \left(     \frac{\diff \vec u}{\diff t} + \vec u \boxtimes  {\mat D}_N \vec u - \nu  {\mat D}^2_N \vec u
    \right) = 0, \qquad \text{where } \mat Z_N = \begin{pmatrix}
        \vec 0 \mid \mat I_{N-1} \mid \vec 0
    \end{pmatrix}
\end{equation*}
is the $(N-1) \times (N+1)$ matrix that extracts the central $N-1$ equations.
Written as a system of ODEs for the interior values, this reads
\begin{equation}
    \begin{pmatrix}
        \frac{\diff}{\diff t} u_N(t,x_1)     \\
        \vdots                    \\
        \frac{\diff}{\diff t} u_N(t,x_{N-1}) \\
    \end{pmatrix}
    = - \mat Z_N
    \left( \vec u \boxtimes  {\mat D}_N \vec u - \nu  {\mat D}^2_N \vec u \right).
\end{equation}
As usual, the natural way to initialize is $\vec u(0) = (u_0 (x_0), \dots, u_0 (x_N))^t$.
We recall \Cref{rem:pointwise evaluation plotting}.
