\section{Nonlinear Elliptic Problems}

\subsection{Semi-linear problems}

Some problems in the applications present ``lower-order'' non-linearities. For example the absorption equation
\begin{equation*}
    -\Delta u + u^q = f
\end{equation*}
is common in chemistry. It is sometimes possible to extend the techniques above to this framework.
Here $q \ge 1$, and for functions that may change sign we interpret $u^q$ as $|u|^{q-1}u$, so that $s \mapsto s^q$ is continuous and non-decreasing.

\subsubsection{Discretisation with piece-wise linear functions}

For simplicity, let $\Omega = (0,1)$ with homogeneous Dirichlet conditions $u(0) = u(1) = 0$. The weak formulation is: find $u \in H^1_0(0,1)$ such that
\begin{equation*}
    \int_0^1 \frac{\diff u}{\diff x} \frac{\diff v}{\diff x} + \int_0^1 u^q v = \int_0^1 f v, \qquad \forall v \in H^1_0(0,1).
\end{equation*}
Since $H^1_0(0,1) \subset C([0,1])$, the term $u^q$ is bounded and the integral makes sense.

Let $V_h = \spannedby\{\varphi_1, \dots, \varphi_{N-1}\}$ be the space of piece-wise linear functions of \Cref{ex:piece-wise linear in d=1} and write $u_h = \sum_{j} U_j \varphi_j$. The key property of the hat functions is $\varphi_j(x_i) = \delta_{ij}$, so that
\begin{equation*}
    u_h(x_i) = U_i .
\end{equation*}
Testing with $v_h = \varphi_i$, the Galerkin method reads
\begin{equation*}
    \sum_{j} a(\varphi_j, \varphi_i)\, U_j + \int_0^1 \Big( \sum_j U_j \varphi_j \Big)^q \varphi_i = \int_0^1 f \varphi_i, \qquad i = 1, \dots, N-1.
\end{equation*}
The non-linear term is an unpleasant function of the coefficients $U_j$: it has to be computed by quadrature in each element, and it has to be recomputed at every iteration of the non-linear solver.

\paragraph{Interpolation of the non-linearity.}
A simple and very popular alternative is to replace $u_h^q$ by its piece-wise linear interpolant $\mathcal I_h(u_h^q)$. Since the nodal values of $u_h^q$ are $u_h(x_j)^q = U_j^q$, we have
\begin{equation*}
    u_h^q \quad \leadsto \quad \mathcal I_h (u_h^q) = \sum_j U_j^q \varphi_j .
\end{equation*}
That is, we look for $u_h \in V_h$ such that
\begin{equation*}
    a(u_h, v_h) + \prodH{\mathcal I_h(u_h^q)}{v_h} = \prodH{f}{v_h}, \qquad \forall v_h \in V_h,
\end{equation*}
where $H = L^2(0,1)$.
Testing with $v_h = \varphi_i$ we obtain
\begin{equation*}
    \sum_j a(\varphi_j, \varphi_i)\, U_j + \sum_j \Big( \int_0^1 \varphi_j \varphi_i \Big) U_j^q = \int_0^1 f \varphi_i, \qquad i = 1, \dots, N-1.
\end{equation*}
With the stiffness matrix $\mat K$ of \Cref{sec:Galerkin elliptic linear algebra} and the mass matrix $\mat M$ of \Cref{sec:Galerkin parabolic linear ODEs}, this is the non-linear algebraic system
\begin{equation}
    \label{eq:Galerkin semilinear algebraic system}
    \mat K \vec u + \mat M \vec u^q = \vec b,
    \qquad
    \vec u = (U_j)_j, \quad \vec u^q \defeq (U_j^q)_j, \quad \vec b = \Big(\int_0^1 f \varphi_i\Big)_i ,
\end{equation}
where the power $\vec u^q$ is taken component-wise. On a uniform mesh, dividing by $h$,
\begin{equation*}
    \frac{-U_{i+1} + 2 U_i - U_{i-1}}{h^2} + \frac{U_{i-1}^q + 4 U_i^q + U_{i+1}^q}{6} = \frac 1 h \int_0^1 f \varphi_i .
\end{equation*}

\subsection{Quasi-linear problems}

Galerkin approximation is used as an analytical tool to prove existence in some problems. For example, the $p$-Laplacian is defined by
\begin{equation*}
    -\Delta_p u = - \diver \left( |\nabla u|^{p-2} \nabla u \right).
\end{equation*}
For $p>1$, define the flux $|\xi|^{p-2}\xi$ to be zero at $\xi=0$. The distributional definition gives
\begin{equation*}
    \langle-\Delta_p u,\varphi\rangle = \int_\Omega |\nabla u|^{p-2} \nabla u \cdot \nabla \varphi, \qquad \forall u\in W^{1,p}(\Omega),\ \varphi\in C_c^\infty(\Omega).
\end{equation*}
This operator is coercive in a non-linear sense.
More generally we can write the quasi-linear elliptic problem
\begin{equation}
    \label{eq:cuasilinear}
    \begin{dcases}
        -\diver \Phi(\nabla u) = f, & \text{in } \Omega           \\
        u = 0,                      & \text{on } \partial \Omega.
    \end{dcases}
\end{equation}
with a flux $\Phi : \R^d \to \R^d$. The $p$-Laplacian corresponds to $\Phi(\xi) = |\xi|^{p-2}\xi$.
For the $p$-Laplacian the reader can find in the literature suitable versions of Céa's lemma for this operator (slightly different for $p \in (1,2)$ and $p\in(2,\infty)$).
However, we want to highlight a common technique known as \emph{convergence by compactness}.
It is more broadly applicable than Céa-type estimates, but it does not guarantee any rate of convergence.
For the following example we follow the presentation in \cite[Section 9.14]{Ciarlet2013}.
\begin{example}[Convergence by compactness for \eqref{eq:cuasilinear}]
    Let $\Omega$ be a bounded Lipschitz domain. Suppose that $\Phi : {\mathbb R^d} \to {\mathbb R^d}$ is continuous,
    monotone in the sense that
    \begin{equation*}
        (\Phi(\xi) - \Phi(\eta) ) \cdot (\xi - \eta) \ge 0, \qquad \forall \xi, \eta \in {\mathbb R^d},
    \end{equation*}
    coercive in $W^{1,p}$ for $p > 1$ in the sense that
    \begin{equation*}
        \Phi(\xi) \cdot \xi \ge A |\xi|^p - B , \qquad \text{for some }A > 0, B \in \R ,
    \end{equation*}
    and has controlled growth, that is,
    \begin{equation}
        \label{eq:quasilinear crecimiento a}
        |\Phi(\xi)| \le C(1+|\xi|^{p-1}).
    \end{equation}
    Finally, let $f \in L^{p'}(\Omega)$ with $p'= \frac{p}{p-1}$.
    Under these hypotheses, we will apply compactness techniques.

    Let $e_i \in W^{1,p}_0 (\Omega)$ be a linearly independent sequence of functions such that the finite-dimensional spaces
    \begin{equation*}
        V_N = \spannedby\{e_1, \dots, e_N\}
    \end{equation*}
    have the following property: for every $v \in W^{1,p}_0(\Omega)$, there exist $v_N \in V_N$ such that $v_N \to v$ in $W^{1,p}_0(\Omega)$.
    \begin{enumerate}
        \item Show that if $v \in W_0^{1,p}(\Omega)$, then $\Phi(\nabla v) \in L^{p'}(\Omega)^d$, so the integral
              $$
                  \int_\Omega \Phi(\nabla v) \cdot \nabla \varphi
              $$
              is finite for all $\varphi \in W_0^{1,p} (\Omega)$.

        \item 	Show that there exists a solution of the weak formulation in $V_N$:
              \begin{equation}
                  \label{eq:Galerkin weak}
                  u_N \in V_N \text{ such that } \qquad \int_\Omega \Phi(\nabla u_N) \cdot \nabla v = \int_\Omega f v, \qquad 	\forall v \in V_N.
              \end{equation}
              To do this, write $u_N = \sum_{i=1}^N  \alpha_i e_i$, and study the function $F: \mathbb R^N \to \mathbb R^N$ given by
              \begin{equation*}
                  F_k (\alpha) = \int_\Omega \Phi\left(\sum_{i=1}^N \alpha_i \nabla e_i\right) \cdot \nabla e_k - \int_\Omega f e_k.
              \end{equation*}
              \begin{enumerate}
                  \item Show that $F$ is monotone, that is,
                        \begin{equation*}
                            (F(\alpha) - F(\beta)) \cdot (\alpha - \beta) \ge 0, \qquad \forall \alpha, \beta \in \mathbb R^N,
                        \end{equation*}
                        and coercive, that is,
                        $$
                            \lim_{R\to \infty} \inf_{|\alpha| \ge R}\frac{ F(\alpha) \cdot \alpha }{ | \alpha | } = \infty.
                        $$

                  \item Apply Brouwer's theorem to deduce that $F$ has at least one zero in the ball $\overline{B(0,R)}$.

                        (Hint: choose $R$ large enough so that $F(\alpha) \cdot \alpha \ge 0$ for $|\alpha| = R$, and study the map $G (\alpha) = - R \frac{F(\alpha)}{|F(\alpha)|}$).


              \end{enumerate}

        \item 	Show that
              \begin{equation*}
                  \| u_N \|_{W_0^{1,p}(\Omega)}  \le C \big(1 + \| f \|_{L^{p'}(\Omega)}^{\frac{1}{p-1}}\big)
              \end{equation*}
              and show that a subsequence of $u_N$ converges weakly in $W_0^{1,p}(\Omega)$ and strongly in $L^p (\Omega)$.

        \item  Prove that if $v_n$ is a sequence in $W_0^{1,p}(\Omega)$, $v \in W_0^{1,p}(\Omega)$ and $\xi \in L^{p'} (\Omega)^d$ are such that
              $$
                  v_n \rightharpoonup v \text{ in } W_0^{1,p} (\Omega), \qquad \Phi(\nabla v_n) \rightharpoonup \xi \text{ in } L^{p'} (\Omega)^d, \qquad \int_\Omega \Phi(\nabla v_n) \cdot \nabla v_n \to \int_\Omega \xi \cdot \nabla v
              $$
              then $\diver \Phi(\nabla v) = \diver \xi$ in the sense of $W^{-1,p'} (\Omega)$, that is
              $$
                  \int_\Omega \Phi(\nabla v) \cdot \nabla \varphi = \int_\Omega \xi \cdot \nabla \varphi, \qquad \forall \varphi \in W_0^{1,p} (\Omega).
              $$

        \item Construct a weak solution of \eqref{eq:cuasilinear}.

        \item Show that ordinary strict monotonicity,
              $(\Phi(\xi)-\Phi(\eta))\cdot(\xi-\eta)>0$ for $\xi\ne\eta$, gives uniqueness for every $p>1$.
              If $p\ge2$, assume the stronger quantitative condition, for some $\theta>0$,
              $$	(\Phi(\xi) - \Phi(\eta) ) \cdot (\xi - \eta) \ge \theta |\xi - \eta|^p , \qquad \forall \xi, \eta \in {\mathbb R^d}, $$
              then the solution is unique. Moreover, show that the map $f \in L^{p'} (\Omega) \mapsto u \in W_0^{1,p}(\Omega)$ is Hölder continuous with exponent $\frac{1}{p-1}$ (in particular Lipschitz continuous if $p = 2$).

        \item In the case  $\Phi(\nabla u) = \tilde\Phi(|\nabla u|) \nabla u$, observe that the solution can be obtained by the direct method of the Calculus of Variations.
    \end{enumerate}
\end{example}

\begin{proof}
    \begin{enumerate}
        \item We have that $|\Phi( \nabla v) |^{\frac p {p-1}} \le C(1 + |\nabla v| ^{p-1}  )^{\frac p {p-1}} \le C_p (1  + |\nabla v|^{p} ) \in L^1 (\Omega)$

        \item We split the proof into two items
              \begin{enumerate}
                  \item
                        Define
                        $$
                            u_\alpha (x) = \sum_i \alpha_i e_i (x).
                        $$
                        Note that
                        $$
                            F_k(\alpha) - F_k(\beta) = \int_\Omega \Big ( \Phi\left(  \nabla u_\alpha   \right)  - \Phi( \nabla u_\beta) \Big) \cdot \nabla e_k.
                        $$
                        Then
                        $$
                            (F_k(\alpha) - F_k(\beta)) (\alpha_k - \beta_k ) = \int_\Omega \Big ( \Phi\left(  \nabla u_\alpha   \right)  - \Phi( \nabla u_\beta) \Big) \cdot (\alpha_k - \beta_k ) \nabla e_k.
                        $$
                        Therefore
                        $$
                            \begin{aligned}
                                (F(\alpha) - F(\beta)) \cdot (\alpha - \beta ) & = \int_\Omega \Big ( \Phi\left(  \nabla u_\alpha   \right)  - \Phi( \nabla u_\beta) \Big) \cdot \nabla u_{\alpha - \beta}       \\
                                                                               & = \int_\Omega \Big ( \Phi\left(  \nabla u_\alpha   \right)  - \Phi( \nabla u_\beta) \Big) \cdot \nabla (u_{\alpha} - u_{\beta}) \\
                                                                               & \ge 0.
                                %				&\ge \alpha \| \nabla (u_\alpha - u_\beta) \|_{L^p}^2 - \beta 
                            \end{aligned}
                        $$

                        Similarly,
                        $$
                            \begin{aligned}
                                F(\alpha) \cdot \alpha & = \int_\Omega \Phi\left(  \nabla u_\alpha   \right) \cdot \nabla u_\alpha - \int_\Omega f u_\alpha \\
                                                       & \ge A \| \nabla u_\alpha \|_{L^p}^p - B |\Omega| - \| f \|_{L^{p'}} \|u_\alpha\|_{L^p}                 \\
                                                       & \ge A C_P^p \| u_\alpha \|_{L^p}^p - B |\Omega| - C\| u_\alpha \|_{L^p},
                            \end{aligned}
                        $$
                        where $C_P$ is the constant in Poincaré's inequality $\|\nabla v\|_{L^p} \ge C_P \|v\|_{L^p}$ for $v \in W^{1,p}_0(\Omega)$.
                        Since all norms in $\R^N$ are equivalent, $\| u_\alpha \|_{L^p} \sim |\alpha|$. Since $p > 1$, the term of order $|\alpha|^p$ dominates and has a positive constant.


                  \item
                        For $R > 0$ sufficiently large, $F(\alpha) \cdot \alpha \ge 0$ if $|\alpha| = R$.

                        If $F$ does not vanish in $\overline B(0,R)$,  $G: \overline B(0,R) \to \overline B(0,R)$ is well defined and continuous. It has a fixed point $\alpha_0$ by Brouwer's fixed point theorem. At such a fixed point, $|\alpha_0| = |G(\alpha_0)| = R$. But, on the other hand,
                        $$
                            R^2 = |\alpha_0|^2 = G(\alpha_0) \cdot \alpha_0 = -R \frac{F(\alpha_0) \cdot \alpha_0}{|F(\alpha_0)|} \le 0,
                        $$
                        a contradiction.

              \end{enumerate}


        \item We have
              $$
                  \begin{aligned}
                      A \| \nabla u_N\|_{L^p}^p  - B |\Omega| \le \int_\Omega \Phi(\nabla u_N) \cdot \nabla u_N = \int_\Omega f u_N \le \| f \|_{L^{p'}} \| u_N \|_{L^p} \le C \| f \|_{L^{p'}} \| \nabla u_N \|_{L^p}.
                  \end{aligned}
              $$
              Thus $\| \nabla u_N \|_{L^p} \le C$, with $C$ independent of $N$. The compactness follows from the Banach--Alaoglu and Rellich--Kondrachov theorems.

        \item This is the so-called Minty's trick. For $\varphi \in W_0^{1,p}(\Omega)$, by monotonicity we have
              $$
                  \int_\Omega (\xi - \Phi(\nabla \varphi)) \cdot \nabla (v - \varphi) = \lim_n \int_\Omega (\Phi(\nabla v_n) - \Phi(\nabla \varphi)) \cdot \nabla (v_n - \varphi) \ge 0.
              $$
              Taking $t > 0$, $\psi \in W_0^{1,p}(\Omega)$ and $\varphi = v - t \psi$ we have
              $$
                  t \int_\Omega (\xi - \Phi(\nabla v - t \nabla \psi)) \cdot \nabla \psi  \ge 0.
              $$
              Therefore
              $$
                  \int_\Omega (\xi - \Phi(\nabla v - t \nabla \psi)) \cdot \nabla \psi  \ge 0.
              $$
              Using \eqref{eq:quasilinear crecimiento a} and the dominated convergence theorem, we can pass to the limit $t \to 0$ to deduce
              $$
                  \int_\Omega (\xi - \Phi(\nabla v )) \cdot \nabla \psi  \ge 0.
              $$
              Since this is also true for $-\psi$, we deduce the statement.

        \item Take a subsequence of $u_N$ that converges weakly in $W^{1,p}_0(\Omega)$ and strongly in $L^p(\Omega)$, and let $v$ be its limit. Let $v_m = u_{N(m)}$ be this subsequence. By the first item, $\Phi(\nabla v_m)$ is bounded in $L^{p'} (\Omega)^d$. Take a further subsequence (not relabelled) such that $\Phi(\nabla v_{m}) \rightharpoonup \xi$ in $L^{p'}(\Omega)^d$.

              Now, for each $\varphi_N \in V_N$ and $m$ large enough so that $N(m) \ge N$, we have $V_N \subset V_{N(m)}$ and hence
              $$
                  \int_\Omega \Phi(\nabla v_{m}) \cdot \nabla \varphi_N = \int_\Omega f \varphi_N.
              $$
              Passing to the limit $m \to \infty$
              $$
                  \int_\Omega \xi \cdot \nabla \varphi_N = \int_\Omega f \varphi_N , \qquad \forall \varphi_N \in V_N.
              $$
              Given $\varphi  \in W_0^{1,p}(\Omega)$ we can take a sequence $\varphi_N \in V_N$ that approximates it strongly. Since $|\xi|, f \in L^{p'} (\Omega)$ we have
              $$
                  \int_\Omega \xi \cdot \nabla \varphi  = \int_\Omega f \varphi, \qquad \forall \varphi \in W_0^{1,p}( \Omega).
              $$
              In particular, taking $\varphi = v \in W_0^{1,p} (\Omega)$ and using the strong convergence in $L^p(\Omega)$,
              $$
                  \int_\Omega \Phi(\nabla v_{m}) \cdot \nabla v_{m} = \int_\Omega f v_{m} \to \int_\Omega f v = \int_\Omega \xi \cdot \nabla v.
              $$

              We apply the previous point to deduce that
              $$
                  \int_\Omega f \varphi  = \int_\Omega \xi \cdot \nabla \varphi = \int_\Omega \Phi(\nabla v) \cdot \nabla \varphi, \qquad \forall \varphi \in W_0^{1,p}(\Omega).
              $$

        \item For equal data, testing with $u-v$ gives $\int_\Omega(\Phi(\nabla u)-\Phi(\nabla v))\cdot(\nabla u-\nabla v)=0$. Strict monotonicity implies $\nabla u=\nabla v$ almost everywhere, and the zero trace gives $u=v$.
              Under the stronger quantitative assumption with $p\ge2$, if $u$ and $v$ have data $f$ and $g$, testing with $u-v$ gives
              $$
                  \begin{aligned}
                      \theta\|\nabla u-\nabla v\|_{L^p}^p
                      &\le\int_\Omega(\Phi(\nabla u)-\Phi(\nabla v))\cdot\nabla(u-v)
                       =\int_\Omega(f-g)(u-v)\\
                      &\le\|f-g\|_{L^{p'}}\|u-v\|_{L^p}
                       \le C\|f-g\|_{L^{p'}}\|\nabla u-\nabla v\|_{L^p}.
                  \end{aligned}
              $$
              So
              $$
                  \| \nabla u - \nabla v \|_{L^p} \le \left( \frac{C}{\theta}  \| f - g \|_{L^{p'}} \right)^{\frac{1}{p-1}}.
              $$
              Taking $f = g$ we recover uniqueness, and in general the solution map $L^{p'} (\Omega) \to W_0^{1,p}(\Omega)$ is Hölder continuous with exponent $\frac 1 {p-1}$.

        \item The energy of the form
              $$
                  \mathcal I (u) = \int_\Omega \Psi(  |\nabla u|  ) - \int_\Omega fu
              $$
              has Lagrangian $L(\xi,s,x) = \Psi(|\xi|) - f(x) s$ with
              $$
                  \frac{\partial L}{\partial \xi_i} = \Psi'(|\xi|) \frac{\xi_i}{|\xi|}.
              $$
              Thus the Euler--Lagrange equation is
              $$
                  f = -\sum_{i=1}^d \frac{\partial}{\partial x_i} \left(   \frac{\Psi'(|\nabla u|)}{|\nabla u|} \frac{\partial u}{\partial x_i}   \right)  = - \diver \left( \frac{\Psi'(|\nabla u|)}{|\nabla u|} \nabla u \right).
              $$
              It suffices that $\Psi'(r) = r \tilde\Phi(r)$.

    \end{enumerate}
\end{proof}

The scheme of the proof of the previous result allows us to prove, in general, the extension of the Lax--Milgram theorem (\Cref{thm:Lax-Milgram}) to the nonlinear framework:
\begin{strongtheorem}[Browder--Minty]
    \label{thm:Browder-Minty}
    Let $X$ be a real reflexive Banach space and let $T : X \to X^*$ be an operator such that
    \begin{enumerate}
        \item $T$ is bounded: it maps bounded sets of $X$ into bounded sets of $X^*$;
        \item $T$ is continuous;
        \item $T$ is monotone: $\langle Tx - Ty, x - y \rangle \ge 0$ for all $x, y \in X$;
        \item $T$ is coercive:
              \begin{equation*}
                  \inf_{\|x\|_X = R} \frac{\langle Tx, x \rangle}{\|x\|_X} \longrightarrow \infty \qquad \text{as } R \to \infty.
              \end{equation*}
    \end{enumerate}
    Then $T$ is surjective, that is, for every $g \in X^*$ there exists $u \in X$ such that $Tu = g$.
\end{strongtheorem}


\begin{remark}
    In general, this construction can be done with any spaces $V_N$ that are useful. The only requirement is to be able to approximate the whole space $V$ as $N \to \infty$. This type of argument is similar to those used to prove the convergence of the Finite Element Method in Numerical Analysis.
\end{remark}


\begin{remark}
    When $\Phi$ does not have natural coercivity on the space $H^1 = W^{1,2}$, one can work in other spaces, for example $W^{1,p}$. This is the case of the $p$-Laplacian operator, where $\Phi(\nabla u) = |\nabla u|^{p-2} \nabla u$. If the situation is even more general, sometimes one can work in the so-called Orlicz spaces, where a Young function $\Psi$ defines the modular $\int_\Omega\Psi(|\nabla u|)$, rather than a norm directly. On the zero-trace space the associated Luxemburg gradient norm is
    \begin{equation*}
        \|\nabla u\|_{L^\Psi}=\inf\left\{\lambda>0:\int_\Omega\Psi\left(\frac{|\nabla u|}{\lambda}\right)\le1\right\}.
    \end{equation*}
\end{remark}
