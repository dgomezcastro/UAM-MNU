\subsection{Variational crimes: Strang's Lemmas}

Problems of the form $-\diver(\mat A(x)\nabla u)= f$ with a variable coefficient $\mat A(x)$ produce Galerkin formulations that are well-posed, but whose integrals are not exactly computable.
When we perform quadrature of the integrals, we arrive at similar formulations
\begin{equation}
    \label{eq:Galerkin quadrature}
    a_h(u_h,v_h) = \ell_h(v_h), \qquad \forall v_h \in V_h,
\end{equation}
where $a_h$ and $\ell_h$ are approximations of $a$ and $\ell$.

There are two well-known approximation estimates
\begin{lemma}[Strang's First Lemma]
    \label{lem:Strang first lemma}
    Assume that $a$ is continuous with constant $M$, that $a, a_h$ are coercive with constant $\alpha^*$, and let $u$ be the solution to \eqref{eq:Galerkin elliptic variational} and $u_h$ be the solution to \eqref{eq:Galerkin quadrature}. Then, we have that
    \begin{align*}
        \| u - u_h \|_V
         & \le \inf_{w_h \in V_h} \left[
            \left( 1 + \frac{M}{\alpha^*}\right) \|u - w_h \|_V + \frac 1 {\alpha^*} \sup_{v_h \in V_h \setminus \{0\}} \frac{|a(w_h, v_h) - a_h(w_h,v_h)|}{\|v_h\|_V}
        \right]                                                                                      \\
         & \qquad + \frac 1 {\alpha^*} \sup_{v_h \in V_h \setminus \{0\}} \frac{|\ell(v_h) - \ell_h(v_h)|}{\|v_h\|_V}.
    \end{align*}
\end{lemma}
The proof is not difficult. We refer to \cite[Theorem 5.5.1]{QuarteroniValli1994}.
\begin{lemma}[Strang's Second Lemma]
    \label{lem:Strang second lemma}
    Under the hypotheses of \Cref{lem:Strang first lemma}
    assume also that $a_h$ is continuous with constant $M$ (and that $a_h(u, \cdot)$ is well defined).
    Then, we have that
    \begin{equation}
        \|u - u_h\|_V \le \left( 1 + \frac{M}{\alpha^*} \right) \inf_{w_h \in V_h} \|u - w_h\|_V + \frac{1}{\alpha^*} \sup_{v_h \in V_h \setminus \{0\}} \frac{|a_h(u, v_h) - \ell_h (v_h)|}{\|v_h\|_V}.
    \end{equation}
\end{lemma}
The proof can be found in \cite[pp.~212--213]{ciarletHandbookNumericalAnalysis1990}.
