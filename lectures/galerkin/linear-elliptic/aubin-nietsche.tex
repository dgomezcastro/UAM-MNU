\subsection{Aubin--Nitsche's duality argument: better convergence in weaker norms}

In \Cref{ex:Galerkin Laplace piecewise linear error estimate} we could be interested in the rate of convergence in $L^2 (\Omega)$, which can be more optimistic.

Throughout this section we assume that $a$ is symmetric, continuous and coercive, so that the energy norm \eqref{eq:energy norm} is a norm equivalent to $\|\cdot\|_V$, and we use the notation of \Cref{sec:Galerkin linear problems Hilbert space} for the operator $A$ associated to $a$ and $H$.
Assume, furthermore, that we can estimate
\begin{equation}
    \label{eq:Galerkin projection estimate in terms of A}
    \inf_{v_h \in V_h} \| v - v_h \|_a \le \omega(h) \| A v \|_{H}, \qquad \forall v \in D_H(A).
\end{equation}
\Cref{ex:cea orthogonality} ensures that $\|u - u_h\|_a \le \omega(h) \| f \|_{H}$. In the weaker norm of $H$ we have the following.
\begin{lemma}[Aubin--Nitsche]
    \label{lem:Aubin-Nietsche}
    Let $f \in H$, $u \in V$ and $u_h \in V_h$ be solutions to
    \begin{align}
        a(u,v)
         & = \prodH{f}{v} , \quad \forall v \in V,
        \qquad
        a(u_h,v_h)
        = \prodH{f}{v_h} , \quad \forall v_h \in V_h,
    \end{align}
    and assume that \eqref{eq:Galerkin projection estimate in terms of A} holds.
    Then, we can estimate
    \begin{equation*}
        \| u - u_h \|_H \le \omega(h)^2 \| f \|_{H}.
    \end{equation*}
\end{lemma}

\begin{proof}
    First, we recall that
    \begin{equation*}
        \| v \|_H = \sup_{g \in H \setminus \{0\}} \frac{\prodH{v}{g}}{\|g\|_H}.
    \end{equation*}
    Let us estimate the right-hand side. Let $g \in H$. We find $w \in V$ and $w_h \in V_h$ such that
    \begin{equation}
        a(w,v) = \prodH{g}{v} , \qquad \forall v \in V,
        \qquad
        a(w_h,v_h) = \prodH{g}{v_h} , \qquad \forall v_h \in V_h.
    \end{equation}
    Using the symmetry of $a$, we operate a little to deduce that
    \begin{align*}
        \prodH{u - u_h} g
         & =a(w,u) - a(w_h,u_h)                                       \\
         & =a(w,u) - \prodH f {w_h}                                   \\
         & =a(w,u) + \prodH g {u_h} - \prodH g {u_h} - \prodH f {w_h} \\
         & =a(w,u) + a(w_h,u_h) - a(w,u_h) - a(u, w_h)                \\
         & = a(w - w_h, u - u_h)
    \end{align*}
    Notice that $A w = g$ and $Au = f$, and hence $\| Aw \|_H = \|g\|_H$ and $\|Au\|_H = \|f\|_H$. We can apply the Cauchy--Schwarz inequality for $a$ and the best-approximation property of \Cref{ex:cea orthogonality} to deduce
    \begin{align*}
        \prodH{u - u_h}{g}
         & = a(w - w_h, u - u_h)                                                                                    \\
         & \le \|w - w_h\|_a \|u - u_h\|_a                                                                          \\
         & \le \left( \inf_{v_h \in V_h } \| w - v_h \|_a \right) \left( \inf_{v_h \in V_h} \| u - v_h \|_a \right) \\
         & \le \omega(h) \| A w \|_H \omega(h) \| A u \|_H                                                          \\
         & = \omega(h)^2 \| f \|_H \| g\|_{H}.
    \end{align*}
    Therefore, we recover the result.
\end{proof}
